	private static List<TResult> Transform<TSource, TResult, TKey>(IEnumerable<TSource> source, Func<TSource, TResult> selector, Func<TResult, TKey> keySelector) where TKey : IComparable<TKey>
	{
		return source.Select(selector).OrderBy(keySelector).ToList();
	}

	private static async Task<List<TResult>> BuildReportAsync<TResult>(IEnumerable<TResult> source)
	{
		await Task.Delay(5).ConfigureAwait(continueOnCapturedContext: false);
		return (from x in source.Select((TResult x, int i) => new
			{
				Index = i,
				Value = x
			})
			where x.Index % 2 == 0
			select x.Value).ToList();
	}

	private static void Dump<T>(string title, IEnumerable<T> items)
